```python
class SpriteTweenSet():
    def __init__(self, prefab):
        self.nid = prefab.nid

        # Load images
        self.full_image = engine.image_load(prefab.full_path)
        self.full_image = self.full_image.convert_alpha()
        width = self.full_image.get_width() // prefab.frame_x
        height = self.full_image.get_height() // prefab.frame_y
        self.images = [engine.subsurface(self.full_image, (i % prefab.frame_x, i // prefab.frame_x, width, height))][:prefab.num_images]

        self.current_idx = 0
        self.current_image = self.images[self.current_idx].copy()

        self.offset = (0, 0)
        self.tweens = []
        self.parent: SpriteTweenSet = None
        self.kill_me = False

    def get_image(self):
        return self.current_image

    def add_tween(self, tween):
        self.tweens.append(tween)

    def update(self):
        self.tweens = [tween for tween in self.tweens if tween.update()]

    def draw(self, surf, pos):
        if self.parent:
            pos = utils.tuple_add(pos, self.parent.offset)
        pos = utils.tuple_add(pos, self.offset)
        surf.blit(self.current_image, pos)
```
